#### Önerilen Asset Deduplication ve Dağılım Doğrulama Mantığı (Kod Örneği):
Sistem B'nin düştüğü mükerrer başlık ve alakasız kümeleme tuzaklarını önlemek için generation pipeline'ında bulunması gereken Python/Pydantic doğrulama mantığı:

```python
import re
from typing import List
from pydantic import BaseModel, field_validator, ValidationInfo

BANNED_UNGROUNDED_CLAIMS = [
    r"\ben iyi\b",
    r"\b%100\b",
    r"\bgaranti(li)?\b",
    r"\b1 numara\b",
    r"\blider\b"
]

class RSAGroupAssets(BaseModel):
    headlines: List[str]
    descriptions: List[str]

    @field_validator("headlines")
    @classmethod
    def validate_headlines(cls, v: List[str]) -> List[str]:
        # 1. Uzunluk ve Count Doğrulaması
        if not (3 <= len(v) <= 15):
            raise ValueError(f"Başlık sayısı 3 ile 15 arasında olmalıdır. Mevcut: {len(v)}")
        
        # 2. Case-Insensitive Deduplication (Sistem B Grup 6 & 10 Hatalarını Engeller)
        normalized_seen = set()
        deduped = []
        for h in v:
            clean = h.strip()
            norm = clean.lower()
            if len(clean) > 30:
                raise ValueError(f"Başlık 30 karakteri aşıyor: '{clean}' ({len(clean)} karakter)")
            if norm not in normalized_seen:
                normalized_seen.add(norm)
                deduped.append(clean)
        
        # 3. İddia / Grounding Denetimi
        for h in deduped:
            for pattern in BANNED_UNGROUNDED_CLAIMS:
                if re.search(pattern, h, re.IGNORECASE):
                    raise ValueError(f"Dayanaksız iddia tespit edildi: '{h}'")
                    
        return deduped

    @field_validator("descriptions")
    @classmethod
    def validate_descriptions(cls, v: List[str]) -> List[str]:
        if not (2 <= len(v) <= 4):
            raise ValueError(f"Açıklama sayısı 2 ile 4 arasında olmalıdır. Mevcut: {len(v)}")
        for d in v:
            clean = d.strip()
            if len(clean) > 90:
                raise ValueError(f"Açıklama 90 karakteri aşıyor: '{clean}' ({len(clean)} karakter)")
            for pattern in BANNED_UNGROUNDED_CLAIMS:
                if re.search(pattern, clean, re.IGNORECASE):
                    raise ValueError(f"Açıklamada dayanaksız iddia tespit edildi: '{clean}'")
        return v
```
